class Payment(models.Model):
    STATUS_CHOICES = [
        ('pending', 'Not Paid'),
        ('completed', 'Paid'),
        ('cancel', 'Cancel'),
    ]

    user = models.ForeignKey(User, on_delete=models.CASCADE, related_name='user_payment',default=None)
    service =  models.ForeignKey(Services, on_delete=models.CASCADE, related_name='user_service',default=None)
    payment_date = models.DateTimeField(auto_now_add=True)
    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default='pending')
    or_number = models.CharField(max_length=100, unique=True, blank=True, null=True)
    description = models.TextField(blank=True, null=True)

    class Meta:
        verbose_name = "Payment"
        verbose_name_plural = "Payments"
        unique_together = ['user','service','status']

    def __str__(self):
        return f"{self.user.username}  - {self.status}"
